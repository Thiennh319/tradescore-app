function populateBollingerTimeframe(target, source):
  percentB, bandwidth, bandwidthSlope, marketMode
  upper, middle, lower  — map 1:1

function populateBollinger(snapshot, bollinger):
  populateBollingerTimeframe(snapshot.bollinger.h1, bollinger.h1)
  populateBollingerTimeframe(snapshot.bollinger.h4, bollinger.h4)

populateRuleAuditSnapshot:
  if (input.bollinger == null) throw 'input.bollinger is required'
  populateBollinger(snapshot, input.bollinger)
