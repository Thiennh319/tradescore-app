validateEntryState(value: string): EntryStateValidationResult
  - Kiểm tra EntryState enum (isEntryState)
  - Kiểm tra ENTRY_STATE_IDS (RuleBook §1)
  - Kiểm tra ENTRY_STATE_DEFINITIONS metadata
  - Error: ESM_002 (unknown), ESM_005 (metadata/rule)

validateTransition(from: string, to: string): EntryStateValidationResult
  - Validate from/to states trước
  - Lookup findTransitionDefinition (locked matrix)
  - validateTransitionMetadata trên row
  - allowed=false → ESM_001
  - Trả transitionDefinition, ruleReference, transitionId, category

validateTransitionMetadata(def: EntryTransitionDefinition): EntryStateValidationResult
  - transitionReason, category, priority, sourceModule, auditLabel, ruleReference
  - Priority phải khớp TRANSITION_CATEGORY_PRIORITY
  - Forbidden → ENTRY_INVALID; allowed → không ENTRY_INVALID
  - Error: ESM_005

EntryStateValidator — namespace { validateEntryState, validateTransition, validateTransitionMetadata }
